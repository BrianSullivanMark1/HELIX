# HELIX V2 — runtime dependencies.
# Install:  pip install -r requirements.txt
# Voice is optional; the app degrades to text-only (and OS voice) if it is absent.

# --- Core (required) ---
PyQt6>=6.7            # the desktop UI (single runtime)
anthropic>=0.40       # the model layer: drives the conversation and the API-based coder
claude-agent-sdk>=0.2 # the subscription brain — runs turns on the user's Claude plan via claude.exe
httpx>=0.27           # HTTP client for the Tripo neural-3D adapter (also a transitive anthropic dep)

# --- Document ingestion (knowledge bases, attached files, and read_file) ---
pypdf>=4.0            # extract text from PDFs
python-docx>=1.1      # extract text from Word (.docx) documents
# Scanned PDFs (no text layer) are OCR'd on-machine: pages render via pdfium, text is read by
# Windows.Media.Ocr — the engine Windows itself ships (fast, offline, nothing to download).
pypdfium2>=5          # render PDF pages to pixels (BSD-licensed, bundles its own pdfium)
winsdk>=1.0.0b10      # WinRT bindings for Windows.Media.Ocr

# --- Vision (attached/pasted images and the camera eye) ---
Pillow>=10            # decode, orient, downscale, re-encode images for the model
# Semantic knowledge search is OPTIONAL and uses the Voyage API over plain HTTP (no extra dependency);
# it is enabled only when a Voyage key is set, and falls back to keyword search otherwise.

# --- 3D models (holograms) ---
# A hologram is an OpenSCAD program (model.scad) compiled by the OpenSCAD command line — a separate
# program HELIX finds on the machine or installs just in time with winget (adapters/openscad_cli.py), NOT a
# Python package, so nothing here is needed to design one. The mesh stack below belonged to the retired
# primitive engine (a declarative model.json turned into assets/model.glb in-process); nothing in helix/
# imports it today. It is kept for now because a Tripo reference GLB and future mesh work may want it —
# drop it deliberately, not by accident. numpy stays regardless: speaker_embed, ocr and voiceid use it.
trimesh>=4.4          # geometry kernel + GLB export (MIT) — currently unused by helix/
numpy>=1.26           # voice-print embeddings, OCR page pixels (and trimesh's array backend)
manifold3d>=2.5       # robust boolean ops (Apache-2.0) — currently unused by helix/
shapely>=2.0          # 2D polygons (BSD) — currently unused by helix/
mapbox-earcut>=1.0    # polygon triangulation (ISC) — currently unused by helix/
# scipy used to sit here for services/materials.py (procedural PBR textures: gaussian_filter). That module
# is gone with the primitive engine and nothing in helix/ imports scipy now, so it is not a dependency —
# tests/test_prompts.py sweeps every module-scope third-party import against this file, and a line for a
# package nobody imports is exactly the drift that sweep exists to catch.

# --- Optional: in-app viewer + GPU orb ---
# Without WebEngine the app still runs whole: main_window falls back to the system browser for built apps
# and holograms, and the orb falls back to its QPainter self. Listed here (like voice) so a fresh install
# gets the full experience instead of a silently reduced one. Deliberately unpinned — PyQt6-WebEngine must
# track the PyQt6 minor above it, and pinning a floor here would fight that resolution.
PyQt6-WebEngine       # the in-app viewer for built apps/holograms (ui/app_viewer.py) + the WebGL orb

# --- The web shell (V3 overhaul): the React face's localhost backend + the app window ---
fastapi>=0.110       # the API + WebSocket server the web face talks to
uvicorn>=0.30        # serves it on 127.0.0.1
pywebview>=5.0       # the desktop window (Edge WebView2) the face lives in
sounddevice>=0.4     # the web voice loop's microphone (PortAudio; Qt-free)

# --- 3D models (holograms): the CAD kernel ---
build123d>=0.7       # the B-rep engine (OCCT) hologram model.py designs compile through — the
                     # worker subprocess (helix/cad/runner.py) is its only importer

# --- Optional: voice ---
edge-tts>=6.1         # free neural text-to-speech
faster-whisper>=1.0   # local, private speech-to-text (audio never leaves the machine)
